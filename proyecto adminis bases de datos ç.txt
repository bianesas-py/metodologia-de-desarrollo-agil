-- CREACION DE LAS TABLAS 
CREATE TABLE cliente(
rut_cliente        NUMBER          PRIMARY KEY,
nombre             VARCHAR2(50)    NOT NULL,
apellido           VARCHAR2(50)    NOT NULL,
correo_electronico VARCHAR2(50)
);

CREATE TABLE producto(
id_producto        NUMBER          PRIMARY KEY,
nombre             VARCHAR2(50)    NOT NULL,
precio             NUMBER          NOT NULL,
stock              NUMBER          NOT NULL,
descripccion       CLOB 
);
CREATE SEQUENCE secuencia_producto
START WITH 5
INCREMENT BY 5
MAXVALUE 9000;

CREATE TABLE carrito(
id_carrito         NUMBER          PRIMARY KEY,
rut_cliente        NUMBER          NOT NULL,
id_producto        NUMBER          NOT NULL,

cantidad           NUMBER          NOT NULL    ,

CONSTRAINT    fk_cliente_carito
FOREIGN KEY   (rut_cliente)
REFERENCES    cliente(rut_cliente),

CONSTRAINT    fk_producto_carito
FOREIGN KEY   (id_producto)
REFERENCES    producto(id_producto)
);
CREATE SEQUENCE secuencia_carrito
START WITH 5
INCREMENT BY 5
MAXVALUE 9000;


CREATE TABLE venta(
id_venta           NUMBER          PRIMARY KEY NOT NULL,
rut_cliente        NUMBER          NOT NULL,
fecha              DATE            DEFAULT SYSDATE,
total              NUMBER ,

CONSTRAINT    fk_venta_cliente
FOREIGN KEY   (rut_cliente)
REFERENCES    cliente(rut_cliente)
);
CREATE SEQUENCE secuencia_venta
START WITH 5
INCREMENT BY 5
MAXVALUE 9000;

CREATE TABLE detalles_ventas(
id_detalles_ven    NUMBER          PRIMARY KEY  NOT NULL,
id_venta           NUMBER          NOT NULL,
id_producto        NUMBER          NOT NULL,
cantidad           NUMBER          NOT NULL ,   


precio_unitario    NUMBER ,

CONSTRAINT    fk_detalles_venta_venta
FOREIGN KEY   (id_venta)
REFERENCES    venta(id_venta),

CONSTRAINT    fk_detalles_venta_producto
FOREIGN KEY   (id_producto)
REFERENCES    producto(id_producto)
);
CREATE SEQUENCE sequencia_detalles_ventas
START WITH 5
INCREMENT BY 5
MAXVALUE 9000;
